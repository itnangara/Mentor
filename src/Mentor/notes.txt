ClassMethod AddAuthor(objString) As %Status
{
    set ID = ""
    set result = {}
    set result = result.%FromJSON(objString)
    Try {
        if (result '= "") {            
            // create a new Author
            Set author = ##class(Mentor.Author).%New()
            Set author.Name = result.Name
            Set author.Email = result.Email
            Set author.Gender = result.Gender

            if (result.BooksValue = "") {
                Set status = author.%Save()
            } else {
                &SQL(
                    Declare SC Cursor 
                    For SELECT ID INTO :ID
                    FROM Mentor.Book
                    WHERE Name = :result.BooksValue
                )
                &SQL(Open SC)

                Quit:(SQLCODE'=0)
                IF SQLCODE<0 { WRITE "SQLCODE error ",SQLCODE," ",%msg  QUIT }
                ELSEIF SQLCODE=100 {WRITE "Author not found"  QUIT}
                
                &SQL(Fetch SC)

                Set bookAuthor = ##class(Mentor.BookAuthor).%New()
                if (ID '= "") {
                    // if ID exists, make a reference to the existing book
                    set book = ##class(Mentor.Book).%OpenId(ID)
                } else {
                    // create a new book
                    Set book = ##class(Mentor.Book).%New()
                    Set book.Name = result.BooksValue
                }
                // link the book and the author using bookAuthor
                Set bookAuthor.Author = author
                Set bookAuthor.Book = book
                Set bookAuthorStatus = bookAuthor.%Save()
            }
        }
    }
    Catch (err) {
        Set %response.Status = 500
        return {"errormessage":"Server Error"}
    }
    return $$$OK
}

Query GetAuthorsWithBooks() As %SQLQuery [ SqlName = AuthorList, SqlProc ]
{
    SELECT
        a.Id AS AuthorId,
        a.Name AS AuthorName,
        b.Name AS BookTitle
    FROM
        Mentor.Author a
    LEFT JOIN
        Mentor.BookAuthor ba ON a.ID = ba.ID
    LEFT JOIN
        Mentor.Book b ON ba.ID = b.ID
}
